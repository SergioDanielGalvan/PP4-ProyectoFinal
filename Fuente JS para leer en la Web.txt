/**
 * Realiza una búsqueda en Buscalibre.
 * @param {string} termino - El título, autor o ISBN a buscar.
 * @returns {Promise<string>} - Promesa con el HTML de la página de resultados.
 */
async function buscarEnBuscalibre(termino) {
    if (!termino || typeof termino !== 'string') {
        throw new Error('El término de búsqueda debe ser una cadena válida.');
    }

    // Construir la URL con el parámetro 'q'
    const baseUrl = 'https://www.buscalibre.com.ar/libros/search/';
    const parametros = new URLSearchParams({ q: termino });
    const urlCompleta = `${baseUrl}?${parametros.toString()}`;

    try {
        const respuesta = await fetch(urlCompleta, {
            method: 'GET',
            headers: {
                // Es posible que el sitio requiera headers como User-Agent o Referer
                // para evitar bloqueos. Ajústalos si es necesario.
                'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36',
                'Accept': 'text/html,application/xhtml+xml,application/xml;q=0.9,*/*;q=0.8',
                'Accept-Language': 'es-ES,es;q=0.9',
            }
        });

        if (!respuesta.ok) {
            throw new Error(`Error en la búsqueda: ${respuesta.status} ${respuesta.statusText}`);
        }

        // La respuesta es HTML, no JSON.
        const html = await respuesta.text();
        return html;

    } catch (error) {
        console.error('Error al buscar en Buscalibre:', error);
        throw error;
    }
}

// Ejemplo de uso: buscar el mismo término que en tu URL
buscarEnBuscalibre('yoshihiro togashi')
    .then(html => {
        console.log('HTML de resultados obtenido. Longitud:', html.length);
        // Aquí podrías parsear el HTML para extraer los libros.
    })
    .catch(error => console.error(error));
